<script setup lang="ts">
import { computed, onUnmounted, ref, watch } from 'vue'
import { useRoute, useRouter } from 'vue-router'
import AsyncState from '@/components/AsyncState.vue'
import { errorMessage } from '@/api/http'
import { useAuthStore } from '@/stores/auth'
import { mockModules } from '@/app/config'
import {
  cachedReport,
  downloadReport,
  getReportEvents,
  getHistory,
  getReportDetails,
  startAnalysis,
} from './analysis-api'
import type { AnalysisReport, AnalysisRun, ReplayDetails, ReplayEvent } from './contracts'
import {
  assertDetailsIdentity,
  describeEvent,
  eventLabels,
  metricLabels,
  moduleLabels,
  numberText,
  reasonLabels,
  roundRows,
  secondsText,
  statusLabels,
} from './report-data'
import ReportChart from './ReportChart.vue'
import RoundMap from './RoundMap.vue'

const route = useRoute(),
  router = useRouter(),
  auth = useAuthStore()
const report = ref<AnalysisReport | null>(null),
  details = ref<ReplayDetails | null>(null)
const loading = ref(true),
  error = ref(''),
  detailsError = ref(''),
  evidenceError = ref(''),
  historyError = ref('')
const downloaded = ref<number | null>(null),
  retrying = ref(false),
  actionError = ref('')
const evidence = ref<Awaited<ReturnType<typeof getReportEvents>> | null>(null),
  history = ref<AnalysisRun[]>([])
const selectedRound = ref(1),
  eventRound = ref('all'),
  eventType = ref('all'),
  search = ref(''),
  eventPage = ref(1)
const focusedEvent = ref<string | null>(null)
let generation = 0
const basic = computed(() => report.value?.report.modules.basic_stats?.metrics)
const rows = computed(() => (details.value ? roundRows(details.value) : []))
const weapons = computed(() =>
  [...(details.value?.statistics.tables?.by_weapon || [])].sort(
    (a, b) => (b.kills ?? -1) - (a.kills ?? -1),
  ),
)
const allEvents = computed(() => (evidence.value?.events || []) as ReplayEvent[])
const eventTypes = computed(() => [...new Set(allEvents.value.map((e) => e.type))].sort())
const filteredEvents = computed(() =>
  allEvents.value
    .filter(
      (e) =>
        (eventRound.value === 'all' || String(e.round_num) === eventRound.value) &&
        (eventType.value === 'all' || e.type === eventType.value) &&
        (!search.value.trim() ||
          `${e.event_id} ${describeEvent(e)}`
            .toLowerCase()
            .includes(search.value.trim().toLowerCase())),
    )
    .sort((a, b) => a.tick - b.tick),
)
const eventPages = computed(() => Math.max(1, Math.ceil(filteredEvents.value.length / 20)))
const visibleEvents = computed(() =>
  filteredEvents.value.slice((eventPage.value - 1) * 20, eventPage.value * 20),
)
const focused = computed(() => allEvents.value.find((e) => e.event_id === focusedEvent.value))
watch([eventRound, eventType, search], () => (eventPage.value = 1))
const metricCards = computed(() => [
  {
    label: '击杀 / 死亡 / 助攻',
    value: [basic.value?.kills, basic.value?.deaths, basic.value?.assists]
      .map((v) => numberText(v, 0))
      .join(' / '),
    note: '本地解析统计',
  },
  { label: 'ADR', value: numberText(basic.value?.adr), note: '平均每回合伤害' },
  {
    label: 'KAST',
    value: basic.value?.kast_pct == null ? '未知' : `${numberText(basic.value.kast_pct)}%`,
    note: '有贡献回合占比',
  },
  {
    label: '爆头率',
    value: basic.value?.headshot_pct == null ? '未知' : `${numberText(basic.value.headshot_pct)}%`,
    note: '爆头击杀占比',
  },
])
const chartBase = {
  backgroundColor: 'transparent',
  color: ['#ed9a60', '#90bcac', '#8caecb'],
  textStyle: { color: '#93989e' },
  tooltip: { trigger: 'axis', renderMode: 'richText' },
  legend: { top: 0, textStyle: { color: '#afb6bc' } },
  grid: { left: 48, right: 45, top: 42, bottom: 45 },

  ...chartBase,
  xAxis: { ...chartBase.xAxis, data: rows.value.map((r) => r.round_num) },
  yAxis: { type: 'value', name: '装备价值 $', splitLine: { lineStyle: { color: '#292f34' } } },

      name: '我方',

      connectNulls: false,
      symbolSize: 4,
      lineStyle: { type: 'dashed' },

  ],
}))
const warningText = computed(() => report.value?.report.warnings || [])
const scoreText = computed(
  () =>
    report.value?.match.teams.map((t) => `${t.name} ${t.score ?? '未知'}`).join(' : ') ||
    '比分未知',
)

  details.value?.key_rounds.find((r) => r.round_num === selectedRound.value),
)
const keyRoundText = computed(() =>
  keyRound.value?.reasons.map((reason) => reasonLabels[reason] || reason).join(' / '),
)

async function load() {
  const current = ++generation,
    id = String(route.params.id)
  loading.value = true
  error.value = ''
  detailsError.value = ''
  evidenceError.value = ''
  historyError.value = ''
  report.value = null
  details.value = null
  evidence.value = null
  history.value = []
  focusedEvent.value = null
  eventRound.value = 'all'
  eventType.value = 'all'
  search.value = ''
  downloaded.value = null
  try {
    const value =
      cachedReport(id) ||
      (await downloadReport(id, (n) => {
        if (current === generation) downloaded.value = n
      }))
    if (current !== generation) return
    report.value = value
    const results = await Promise.allSettled([
      getReportEvents(value.package_id),
      getHistory(value.match_id),
      mockModules.replay ? Promise.resolve(null) : getReportDetails(id),
    ])
    if (current !== generation) return
    if (
      results[0].status === 'fulfilled' &&
      results[0].value.match_id === value.match_id &&
      results[0].value.package_id === value.package_id
    )
      evidence.value = results[0].value
    else if (results[0].status === 'fulfilled') evidenceError.value = '事件证据与当前报告身份不一致'
    else evidenceError.value = errorMessage(results[0].reason)
    if (results[1].status === 'fulfilled') history.value = results[1].value.items
    else historyError.value = errorMessage(results[1].reason)
    if (results[2].status === 'fulfilled' && results[2].value) {
      try {
        assertDetailsIdentity(results[2].value, value)
        details.value = results[2].value
        selectedRound.value = rows.value[0]?.round_num ?? 1
      } catch (e) {
        detailsError.value = errorMessage(e)
      }
    } else if (results[2].status === 'rejected')
      detailsError.value = errorMessage(results[2].reason)
  } catch (e) {
    if (current === generation) error.value = errorMessage(e)
  } finally {
    if (current === generation) loading.value = false
  }
}
function selectRound(round: number) {
  if (rows.value.some((r) => r.round_num === round)) selectedRound.value = round
}
function showEvidence(ids: string[]) {
  const event = allEvents.value.find((e) => ids.includes(e.event_id))
  if (!event) {
    actionError.value = '此条依据暂未读取，请重试证据加载'
    return
  }
  eventRound.value = 'all'
  eventType.value = 'all'
  search.value = event.event_id
  focusedEvent.value = event.event_id
  document.getElementById('report-events')?.scrollIntoView({ behavior: 'smooth', block: 'start' })
}
async function reanalyze() {
  if (!report.value) return
  retrying.value = true
  actionError.value = ''
  try {
    const run = await startAnalysis(report.value, crypto.randomUUID())
    sessionStorage.setItem(`csdm.pending-analysis.${auth.user?.id || 'guest'}`, run.analysis_id)
    await router.push('/replays/new?view=cloud')
  } catch (e) {
    actionError.value = errorMessage(e)
  } finally {
    retrying.value = false
  }
}
function exportReport() {
  if (!report.value) return
  const url = URL.createObjectURL(
    new Blob([JSON.stringify({ report: report.value, details: details.value }, null, 2)], {
      type: 'application/json',
    }),
  )
  const link = document.createElement('a')
  link.href = url
  link.download = `replay-${report.value.analysis_id}.json`
  link.click()
  setTimeout(() => URL.revokeObjectURL(url), 1000)
}
watch(() => [route.params.id, auth.user?.id], load, { immediate: true })
onUnmounted(() => generation++)
</script>
